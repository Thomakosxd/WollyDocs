<AccordionGroup>
  <Accordion title="/invitetracking user [user]">
    Check invite statistics for yourself or another member
  </Accordion>

  <Accordion title="/invitetracking leaderboard">
    Show top inviters in the server (with Refresh button)
  </Accordion>

  <Accordion title="/invitetracking add <user> <amount>">
    Add bonus invites to a user
  </Accordion>

  <Accordion title="/invitetracking remove <user> <amount>">
    Remove bonus invites from a user
  </Accordion>

  <Accordion title="/invitetracking set <user> <amount>">
    Set exact bonus invites for a user
  </Accordion>

  <Accordion title="/invitetracking help">
    Show all Invite Tracking commands
  </Accordion>
</AccordionGroup>
